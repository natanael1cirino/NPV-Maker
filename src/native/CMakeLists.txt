cmake_minimum_required(VERSION 3.21)
project(NPVMakerRuntime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(RED4EXT_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/vendor/RED4ext.SDK" CACHE PATH "RED4ext SDK checkout")
add_library(NPVMakerRuntime SHARED Main.cpp)
target_include_directories(NPVMakerRuntime PRIVATE "${RED4EXT_SDK}/include")
target_compile_definitions(NPVMakerRuntime PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(NPVMakerRuntime PRIVATE /W4 /WX /utf-8)

option(NPV_BUILD_LAUNCHER_TESTS "Build isolated launcher lifecycle test tools" ON)
if(NPV_BUILD_LAUNCHER_TESTS)
  add_executable(NPVMakerLauncherHost ../../tests/native/LauncherHost.cpp)
  target_include_directories(NPVMakerLauncherHost PRIVATE "${RED4EXT_SDK}/include")
  target_compile_definitions(NPVMakerLauncherHost PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  add_executable(NPVMakerTestConverter ../../tests/native/TestConverter.cpp)
  target_compile_definitions(NPVMakerTestConverter PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
